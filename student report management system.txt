#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define F "students.txt"

struct S { int r; char n[50]; float m; };
struct U { char u[20], p[20], role[20]; } users[] = {
    {"admin","admin123","admin"},
    {"staff1","staff123","staff"},
    {"guest","guest","guest"}
};

char role[20];

void init() {
    FILE *fp = fopen(F,"r");
    if (fp) { fclose(fp); return; }
    fp = fopen(F,"w");
    fprintf(fp,"1 John 85\n2 Mary 90\n");
    fclose(fp);
}

int login() {
    char u[20], p[20];
    scanf("%s%s",u,p);
    for(int i=0;i<3;i++)
        if(!strcmp(u,users[i].u)&&!strcmp(p,users[i].p)){
            strcpy(role,users[i].role); return 1;
        }
    return 0;
}

void add(){ struct S s; FILE *fp=fopen(F,"a");
    scanf("%d%s%f",&s.r,s.n,&s.m);
    fprintf(fp,"%d %s %.2f\n",s.r,s.n,s.m); fclose(fp); }

void show(){ struct S s; FILE *fp=fopen(F,"r");
    while(fscanf(fp,"%d%s%f",&s.r,s.n,&s.m)!=EOF)
        printf("%d %s %.2f\n",s.r,s.n,s.m);
    fclose(fp); }

void search(){ int r,f=0; struct S s; scanf("%d",&r);
    FILE *fp=fopen(F,"r");
    while(fscanf(fp,"%d%s%f",&s.r,s.n,&s.m)!=EOF)
        if(s.r==r){ printf("%d %s %.2f\n",s.r,s.n,s.m); f=1; }
    if(!f) printf("Not found\n"); fclose(fp); }

void update(){ int r,f=0; struct S s; scanf("%d",&r);
    FILE *fp=fopen(F,"r"),*t=fopen("t.txt","w");
    while(fscanf(fp,"%d%s%f",&s.r,s.n,&s.m)!=EOF){
        if(s.r==r){ scanf("%s%f",s.n,&s.m); f=1; }
        fprintf(t,"%d %s %.2f\n",s.r,s.n,s.m);
    }
    fclose(fp); fclose(t); remove(F); rename("t.txt",F);
    if(!f) printf("Not found\n"); }

void del(){ int r,f=0; struct S s; scanf("%d",&r);
    FILE *fp=fopen(F,"r"),*t=fopen("t.txt","w");
    while(fscanf(fp,"%d%s%f",&s.r,s.n,&s.m)!=EOF){
        if(s.r==r){ f=1; continue; }
        fprintf(t,"%d %s %.2f\n",s.r,s.n,s.m);
    }
    fclose(fp); fclose(t); remove(F); rename("t.txt",F);
    if(!f) printf("Not found\n"); }

// main
int main(){
    init();
    if(!login()){ printf("Login failed\n"); return 0; }

    int c;
    while(scanf("%d",&c)){
        if(!strcmp(role,"admin")){
            if(c==1)add(); else if(c==2)show();
            else if(c==3)search(); else if(c==4)update();
            else if(c==5)del(); else break;
        }
        else if(!strcmp(role,"staff")){
            if(c==1)show(); else if(c==2)search(); else break;
        }
        else{
            if(c==1)show(); else break;
        }
    }
}

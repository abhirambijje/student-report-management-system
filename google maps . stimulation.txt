#include <iostream>
#include <vector>
#include <stack>
#include <map>
using namespace std;

// Add bidirectional road
void addRoad(map<string, vector<string>>& g, string a, string b) {
    g[a].push_back(b);
    g[b].push_back(a);
}

// DFS path finder
bool dfs(map<string, vector<string>>& g, string s, string d, vector<string>& path) {
    map<string, bool> vis;
    map<string, string> parent;
    stack<string> st;

    st.push(s);
    vis[s] = true;

    while (!st.empty()) {
        string cur = st.top(); st.pop();

        if (cur == d) break;

        for (auto& nb : g[cur]) {
            if (!vis[nb]) {
                vis[nb] = true;
                parent[nb] = cur;
                st.push(nb);
            }
        }
    }

    if (!vis[d]) return false;

    // Reconstruct path
    for (string at = d; at != s; at = parent[at])
        path.insert(path.begin(), at);
    path.insert(path.begin(), s);

    return true;
}

int main() {
    map<string, vector<string>> graph;
    int n;
    cin >> n;

    for (int i = 0; i < n; i++) {
        string u, v;
        cin >> u >> v;
        addRoad(graph, u, v);
    }

    string src, dest;
    cin >> src >> dest;

    vector<string> path;

    if (dfs(graph, src, dest, path)) {
        for (auto& p : path) cout << p << " -> ";
        cout << "END\n";
    } else {
        cout << "No route found\n";
    }
}
